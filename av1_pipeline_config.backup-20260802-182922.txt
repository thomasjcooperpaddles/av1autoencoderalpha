# Which encoder does the work.
#   auto    test the GPU at startup, fall back to the CPU if it cannot
#           encode AV1. This is the right setting for almost everyone.
#   nvenc   force the GPU. The run stops if it will not encode AV1.
#   svtav1  force the CPU encoder and skip the GPU test.
# Only Ada (RTX 40) and newer NVIDIA cards can encode AV1 at all. On a
# card that can, the GPU encoder still needs an ffmpeg build and a driver
# that agree on the NVENC API version. When they do not, auto keeps the
# batch moving on the CPU and switches back to the GPU on its own as
# soon as the mismatch is resolved.

# CPU encoder speed, 0 slowest and best through 13 fastest and worst.
# 4 to 8 is the useful range. 6 is a good starting point. Every step
# up roughly halves the encode time and costs a little quality.

# Shifts every computed quality number. Negative means better quality
# and larger files; each step is worth roughly 12 percent bitrate, so
# -2 is about +26 percent and -4 about +59 percent.
#
# This is the only quality lever NVENC has left. Preset, tune, lookahead
# and adaptive quantisation are already at their maximum settings, so
# there is nothing else to turn up: the remaining choice is bitrate.
# If you want genuinely better quality per byte rather than more bytes,
# set encoder = svtav1 instead, which is measurably more efficient than
# any hardware encoder but takes hours per film rather than minutes.
cq_offset = 0

# When deband runs.
#   auto    only on sources under 10-bit, where 8-bit quantisation
#           banding actually exists to be fixed. Default.
#   always  on everything. Costs real CPU at 4K, and on a clean 10-bit
#           master it adds dither rather than removing steps.
#   off     never.
# This is a quality choice, not a speed one. If you see banding in dark
# gradients on a 4K encode, set this to always and compare.
deband_mode = auto

# Hardware video decode. auto tests it at startup and uses it if the

# What happens to the audio track.
#
#   auto    keep the original stream untouched when it is already a
#           codec every player handles (AC3, E-AC3, AAC, DTS, TrueHD,
#           FLAC, MP3, Opus, PCM), and convert anything more awkward to
#           Opus. This is the default and it loses nothing on a normal
#           DVD or Blu-ray rip.
#   copy    never re-encode. Truest to the source. A very old or unusual
#           codec may then need transcoding at playback time.
#   opus    always convert to Opus. Best quality for the size of any
#           current lossy codec, and handles surround properly.
#   aac     always convert to AAC. Only worth it for a player that
#           refuses Opus.
#
# Re-encoding audio that is already lossy loses a little every time, so
# copying is the honest answer whenever the source codec is fine.
audio_mode = auto

# Set to 1 to fold surround down to stereo. Off by default, so a 5.1
# track stays 5.1. Turning this on forces a re-encode, because filters
# cannot be applied to a copied stream.
audio_downmix = 0

# Set to 1 to run a noise reducer over badly starved audio. Off by
# default. It only engages below about 64 kbps a channel, so a normal
# AC3 5.1 track at 448k is never touched. Leave it off unless you are
# working with genuinely poor sources; it smears detail on anything else.
audio_denoise = 0

# ---------------------------------------------------------------------
# The 1080p standard
# ---------------------------------------------------------------------
# The target is 1080 lines of ACTUAL PICTURE, measured after black bars
# are cropped away. Width follows whatever aspect the film is in, so a
# 2.39:1 scope film lands at 2580x1080 and a 16:9 one at 1920x1080.
# Nothing is ever stored as black bars.
#
# Black bars are found by sampling across the whole file and taking the
# shape the film spends most of its time in. Top, bottom, side, and
# postage-stamp bars are all handled. If a film genuinely changes aspect
# part way through, the dominant ratio wins and the log says so.
#
# What happens to a file then follows from the cropped picture height:
#   under 1080 lines   upscaled, and eligible for the AI upscaler
#   exactly 1080       left alone, with a contrast adaptive sharpen
#   over 1080          downscaled, which oversamples and is already sharp
# A 1080p file with bars has less than 1080 lines of real picture, so it
# gets upscaled like any other short source.

# Contrast Adaptive Sharpen strength, 0 to 1. Only used where no rescale
# happens, since a resize brings its own sharpness change. This is the
# FidelityFX CAS algorithm, but ffmpeg runs it on the CPU, so it works
# the same on any graphics card. 0 turns it off.
cas_strength = 0.30

# Saturation lift for standard definition sources, 1.0 is unchanged.
# This is a deliberate look change rather than a restoration, so keep it
# modest. Only applied to sources at or below 720 lines.
sd_saturation = 1.06

# How much of the AI upscale to keep, 0 to 100.
#
# Real-ESRGAN looks at one frame at a time and has no memory of the
# frame before it, so it invents slightly different detail every frame.
# On a still background that reads as crawling or shimmering. Tools like
# Topaz avoid it by feeding the model several frames at once, which is a
# property of the model and cannot be added here.
#
# What this does instead is blend the model output back towards a plain
# lanczos upscale. 100 is the raw model with all its shimmer, 0 is no AI
# at all, 50 keeps half the invented detail and roughly halves the
# flicker with it. Lower this further if backgrounds still crawl.
sr_strength = 50

# Temporal stabiliser applied after the blend. Averages each pixel
# across neighbouring frames only while the difference stays small, so
# still backgrounds settle while anything actually moving passes through
# untouched. Set to 0 to turn it off.
sr_stabilise = 1

# Fast local folder for the upscaler's frame files. Leave blank to use

# Set to 1 to skip Real-ESRGAN entirely and upscale with lanczos.

no_sr = 0

# Set to 1 to expand SDR sources into HDR10 color. Off by default

sdr_to_hdr = 0
