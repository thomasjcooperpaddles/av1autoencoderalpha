# ---------------------------------------------------------------------
# Audio conform  -  OFF, and this is deliberate
# ---------------------------------------------------------------------
# This measured where the rendered audio sat against the master and
# rewrote it to match. It damaged two finished films on 2026-08-11 and
# is disabled.
#
# The flaw is structural, not a tuning problem: a measurement that is
# wrong ONCE destroys a file that was right. Correlating a heavily
# processed stereo downmix against a raw surround master is guesswork,
# and on self-similar film audio it guesses wrong. It read The Simpsons

# Target resolution

# How many lines of ACTUAL PICTURE to aim for. Bars are never counted,
# so a letterboxed film is measured on the picture inside them.
#   1080   the standard, and the default
#   2160   4K, opt-in

# What counts as "close to 4K", and so eligible for a 4K master.
# Measured on the stored WIDTH rather than the height, because width
# survives letterboxing: a 2.39:1 UHD is 3840x1608 once the bars come

# Real picture within this percentage of 2160 lines is brought UP to

# Size ceiling for a 4K master, in GB. SOFT on purpose. The target is
# the smaller of the source's own size and this number, so a 6 GB web-dl
# aims at 6 GB rather than being inflated to fill the allowance.
#
# If holding the quality floor below still predicts an oversized file,
# THE FILE WINS and the log says so loudly. Picture quality outranks
# size. For scale: 10 GB on a 100 minute film is about 0.055 bits per
# pixel, against the 0.194 the 1080p standard spends - roughly a third.
uhd_max_gb = 10.0

# The quality floor, expressed as the worst CQ the size budget may drive
# to. CQ rises to fit the budget and stops here.
#

# Artifact repair for damaged 4K sources.

#   off     never
#   weak    always, gentle
#   strong  always, aggressive

# ---------------------------------------------------------------------
# The 1080p standard
# ---------------------------------------------------------------------
# The target is 1080 lines of ACTUAL PICTURE, measured after black bars
# are cropped away. Width follows whatever aspect the film is in, so a
# 2.39:1 scope film lands at 2580x1080 and a 16:9 one at 1920x1080.
# Nothing is ever stored as black bars.
#
# Black bars are found by sampling across the whole file and taking the
# shape the film spends most of its time in. Top, bottom, side, and
# postage-stamp bars are all handled. If a film genuinely changes aspect
# part way through, the dominant ratio wins and the log says so.
#
# What happens to a file then follows from the cropped picture height:
#   under 1080 lines   upscaled, and eligible for the AI upscaler
#   exactly 1080       left alone, with a contrast adaptive sharpen
#   over 1080          downscaled, which oversamples and is already sharp
# A 1080p file with bars has less than 1080 lines of real picture, so it
# gets upscaled like any other short source.

# Contrast Adaptive Sharpen strength, 0 to 1. Only used where no rescale
# happens, since a resize brings its own sharpness change. This is the
# FidelityFX CAS algorithm, but ffmpeg runs it on the CPU, so it works
# the same on any graphics card. 0 turns it off.
cas_strength = 0.30

# Saturation lift for standard definition sources, 1.0 is unchanged.
# This is a deliberate look change rather than a restoration, so keep it
# modest. Only applied to sources at or below 720 lines.
sd_saturation = 1.06

# How much of the AI upscale to keep, 0 to 100.
#
# Real-ESRGAN looks at one frame at a time and has no memory of the
# frame before it, so it invents slightly different detail every frame.
# On a still background that reads as crawling or shimmering. Tools like
# Topaz avoid it by feeding the model several frames at once, which is a
# property of the model and cannot be added here.
#
# What this does instead is blend the model output back towards a plain
# lanczos upscale. 100 is the raw model with all its shimmer, 0 is no AI
# at all, 50 keeps half the invented detail and roughly halves the
# flicker with it. Lower this further if backgrounds still crawl.
sr_strength = 50

# Temporal stabiliser applied after the blend. Averages each pixel
# across neighbouring frames only while the difference stays small, so
# still backgrounds settle while anything actually moving passes through
# untouched. Set to 0 to turn it off.
sr_stabilise = 1

# destination's .tmp folder, which is the old behaviour and is slower.
scratch_auto = 1

# A disk needs at least this much free space, in GB, to be considered
# for scratch at all. The real requirement is worked out per run from
# the largest file in the queue; this is the floor beneath that.
scratch_min_free_gb = 50

# Upscale factor for the AI model.
#   0  pick the smallest the model has that clears 1080 lines
#   2  force x2
# The cost is quadratic. A 528 line film at x4 becomes 2112 lines and
# then most of those pixels are thrown away scaling back down to 1080;
# x2 lands at 1056 and the last 2 percent is a near-free resize. Four
# times fewer pixels to compute.
#
# NOTE: only realesr-animevideov3 has x2 and x3, which is why it is now
# the only model offered - the 4x-only models made this setting a lie.
sr_scale = 2

# Tile size handed to the upscaler, in pixels.
#   0  work it out from the video memory actually free at the time
#   N  force that tile
#
# Leave this at 0. Memory cost goes as tile SQUARED and again as scale
# SQUARED, so a fixed number that is fine at x2 can be several times
# over the card at x4. A 1024 tile at x4 is about 11 GB on its own.
sr_tile = 0

# How much of the card a run may aim at in total, in MB, and how much
# to hold back.
#
# The upscaler does NOT have the card to itself: the encode of the
# previous file runs alongside it and was measured at 2.3 to 3.4 GB,
# and the desktop wants some too. The reserve is what keeps that from
# becoming a crash. The tile is then sized from whatever is genuinely
# free at the moment the upscale starts, less the reserve, so a busy
# card simply gets a smaller tile instead of falling over.
#
# If it still runs out, the pipeline halves the tile and retries rather
# than failing the file, and the log says where it settled.
sr_vram_mb = 15000
# Held back for the NVENC session that encodes each upscaled chunk while
# the model is still resident. Measured at 2.3 to 3.4 GB, so 3400.
sr_vram_reserve_mb = 3400

# ---------------------------------------------------------------------
# GPU safety
# ---------------------------------------------------------------------
# HARD ceiling on TOTAL video memory in use, as a percentage of the card.
# This is the line that must never be crossed. Crossing it does not fail
# a file politely - it resets the display driver, and on 2026-08-06 it
# took the whole machine down: six nvlddmkm errors at 18:21, then an
# unexpected shutdown.
#
# 98 leaves the desktop compositor and anything else running room. Set
# lower if you use the machine while it works. 100 is refused.
gpu_max_pct = 98

# Largest tile the upscaler may be given, whatever the arithmetic says.
# 896 at x4 was MEASURED safe alongside a running encode, at about
# 8.3 GB. The budget picked 992 on 2026-08-06 and the driver fell over
# 31 minutes later. The underlying cost model comes from ncnn's own
# auto-select thresholds, which are heuristics rather than measured
# allocations, so it is not trusted past what has actually been tried.
sr_tile_max = 896

# Seconds to let the driver hand video memory back after a process is
# killed. nvidia-smi keeps reporting the old figure for a moment, and
# sizing the next job against that stale number turns one over-commit
# into two.
gpu_settle_sec = 5

# How long to wait for video memory before giving up on a file and
# moving on to the next. Never blocks forever; never barges in either.
gpu_wait_sec = 300

# ---------------------------------------------------------------------
# Load ceilings, after three hard machine freezes
# ---------------------------------------------------------------------
# On 2026-08-06, and twice on 2026-08-09, the whole machine locked up.
# Every one of them recorded Kernel-Power 41 with BugcheckCode=0,
# PowerButtonTimestamp=0, no crash dump and no WHEA hardware error.
# That combination means Windows did NOT crash - there was no blue
# screen and nobody held the power button. The kernel simply stopped
# running, which is what heat, power delivery or memory instability do.
# Software cannot fix that. What it CAN do is stop standing on the
# limit, which is what everything below is for.
#
# Temperature the card must come back under before more work is added.
# 0 disables the check.
gpu_temp_max_c = 80
gpu_cool_sec = 15

# Seconds to idle between upscaler chunks.
#
# Was 2.0, from when the lockups were thought to be a load problem. With
# the chunk encode now running alongside the model instead of after it, a
# chunk is about 6 seconds, so 2.0 was 20 percent of it - a quarter of an
# hour of deliberate idling per film. The lockups were the 2022
# ncnn/Vulkan binary on Blackwell, not load.
#
# gpu_temp_max_c above is the real protection: it refuses work when the
# card is actually hot instead of pausing on a timer regardless.
sr_chunk_pause_sec = 0.0

# Prepare the NEXT file (probe, crop analysis, whisper, translation)
# while this one is upscaling. 1 = yes, 0 = hold it back until the
# upscale finishes.
#
# This was 0 for a long time because whisper stacked on the Vulkan
# upscaler was the heaviest load the pipeline could produce, and the
# machine locked up doing exactly that on 2026-08-09. The upscaler is
# CUDA now and the lockup was traced to the Vulkan binary rather than to
# load, so it is back on.
#
# IF THE MACHINE LOCKS UP DURING AN UPSCALE, SET THIS TO 0 FIRST.
sr_prefetch_during = 1

# Threads for load : process : save. The tool's own default is 1:2:2,
# and with thousands of small PNG frames one decode thread leaves the
# card waiting on the disk instead of computing.
sr_threads = 4:4:4

# Set to 1 to skip Real-ESRGAN entirely and upscale with lanczos.
# Much faster, noticeably softer. Leave at 0 for normal runs.
no_sr = 0

# Set to 1 to expand SDR sources into HDR10 color. Off by default
# because it changes the look of every SDR file in the batch.
sdr_to_hdr = 0

# Saturation gain applied during that SDR->HDR expansion.
#
# HARD CEILING 1.30. This path may never boost saturation by more than
# 30 percent, whatever is written here; anything higher is clamped, and
# the floor is 0.25.
#
# 1.0 is neutral. MEASURED on Son of Batman with source and output both
# at 10-bit, the expansion leaves encoded chroma at 0.99x - it does not
# boost the data. What changes is that the file is tagged bt2020/PQ, so
# a display renders it in HDR mode, where SDR-derived material looks
# brighter and more saturated than the SDR original. Same data,
# different rendering. Set this BELOW 1.0 to tame that.
#
# Measured response of the filter, relative chroma at each gain:
#   1.00 = 1.00x      0.60 = 0.57x      0.40 = 0.36x
#   0.50 = 0.47x      0.35 = 0.31x      0.30 = 0.26x
#
# Only applies when sdr_to_hdr is on. Ignored otherwise.
sdr_to_hdr_saturation = 1.0
